# 安装系统原生nginx
dnf install -y nginx

# 放行SELinux【重中之重，不然代理docker内部服务出现502】
setsebool -P httpd_can_network_connect 1

# 防火墙开放80/443（非必须）
firewall-cmd --add-service=http --permanent
firewall-cmd --add-service=https --permanent
firewall-cmd --reload

# 启动&开机自启
systemctl start nginx
systemctl enable nginx

# 查看版本
nginx -v

# 检查状态
systemctl status nginx

# 设置配置文件
vim /etc/nginx/nginx.conf

server {
    listen 80;
    server_name: aide-test.cndrealty.com,
    return 30l https://$host$request uri;
}

server{
    listen 443 ssl;
    server_name aide-test.cndrealty.com;
    ssl_protocols TLSv1.2 TLSv1.3;
    ssl_ciphers 'TLS_AES_128_GCM_SHA256:TLS_AES_256_GCM_SHA384:ECDHE-RSA-AES128-GCM-SHA256:ECDHE-RSA-AES256-GCM-SHA384:ECDHE-RSA-CHACHA20-POLY1305';
    ssl_certificate "/etc/nginx/cert/cndrealty.com.pem";
    ssl_certificate_key "/etc/nginx/cert/cndrealty.com.key";

    # 处理dify
    location /dify-api/ {
        rewrite ^/dify-api/(.*)$ /$1 break;
        proxy_pass http://10.194.73.11;    
    }

    # 处理api
    location /api/ {
        rewrite ^/api/(.*)$ /api/$1 break;
        proxy_pass http://127.0.0.1:7070;       
    }
    
    # 文件系统
    location /minio/ {
        rewrite ^/minio/(.*)$ /$1 break;
        proxy_pass http://10.194.107.189:9000;
    }
    
    # 消息队列
    location /rabbitmq/ {
        rewrite ^/rabbitmq/(.*)$ /$1 break;
        proxy pass http://10.194.107.189:15672;
    }
    
    # 单点登录
    location /sso/login {
        rewrite ^(.*)$ https://authdev.cndrealty.com:8443/oauth2.0/authorize?client_id=AUTH_FC_SYZHSC&grant_type=authorization_code&redirect_uri=https%3A%2F%2Fvmall.jfsytest.com%2Flogin&response_type=code permanent;
    }
    
    # 处理前端请求
    location /prod-api/ {
        rewrite ^/prod-api/(.*)$ /$1 break;
        proxy_pass http://127.0.0.1:7070;       
    }
    
    location / {
        root   /mnt/ai-community/dist;
        index  index.html index.htm;
        try_files $uri $uri/ /index.html;  
    } 
}


7、刷新配置
sudo systemctl reload nginx
